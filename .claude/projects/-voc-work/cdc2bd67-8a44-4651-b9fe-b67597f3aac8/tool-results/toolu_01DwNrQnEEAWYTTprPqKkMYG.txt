     1→import json
     2→from pathlib import Path
     3→from typing import Any, TextIO
     4→
     5→
     6→class Tracer:
     7→    """Append-only JSONL tracer. One JSON object per call to `write`."""
     8→
     9→    def __init__(self, path: Path) -> None:
    10→        self.path = path
    11→        path.parent.mkdir(parents=True, exist_ok=True)
    12→        self._fh: TextIO = path.open("a", encoding="utf-8")
    13→
    14→    def write(self, event: dict[str, Any]) -> None:
    15→        self._fh.write(json.dumps(event, ensure_ascii=False, default=str) + "\n")
    16→        self._fh.flush()
    17→
    18→    def close(self) -> None:
    19→        self._fh.close()
    20→
    21→    def __enter__(self) -> "Tracer":
    22→        return self
    23→
    24→    def __exit__(self, *exc: object) -> None:
    25→        self.close()
    26→
    27→
    28→class NullTracer(Tracer):
    29→    """In-memory tracer for tests; never touches the filesystem."""

    31→    def __init__(self) -> None:
    32→        self.events: list[dict[str, Any]] = []
