   158→def make_executor(session: ClaimSession) -> Executor:
   159→    """Return a ToolExecutor callable bound to this session."""
   160→
   161→    def execute(name: str, tool_input: dict[str, Any]) -> str:
   162→        handler = _DISPATCH.get(name)
   163→        if handler is None:
   164→            return _err("permanent", False, f"unknown tool: {name}")
   165→        try:
   166→            return handler(session, tool_input)
   167→        except Exception as exc:
   168→            # Defensive: anything raised inside a handler becomes a graceful error,
   169→            # never crashes the loop.
   170→            return _err("transient", True, f"{type(exc).__name__}: {exc}")
   171→
   172→    return execute
