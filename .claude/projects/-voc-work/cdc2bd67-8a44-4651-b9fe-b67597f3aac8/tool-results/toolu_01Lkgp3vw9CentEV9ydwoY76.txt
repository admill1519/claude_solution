   250→def make_executor(session: ClaimSession) -> Executor:
   251→    """Return a ToolExecutor callable bound to this session."""
   252→
   253→    def execute(name: str, tool_input: dict[str, Any]) -> str:
   254→        handler = _DISPATCH.get(name)
   255→        if handler is None:
   256→            return _err("permanent", False, f"unknown tool: {name}")
   257→        try:
   258→            return handler(session, tool_input)
   259→        except Exception as exc:
   260→            # Defensive: anything raised inside a handler becomes a graceful error,
   261→            # never crashes the loop.
   262→            return _err("transient", True, f"{type(exc).__name__}: {exc}")
   263→
   264→    return execute
